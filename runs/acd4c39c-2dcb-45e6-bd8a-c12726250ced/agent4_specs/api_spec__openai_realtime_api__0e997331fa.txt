SERVICE: OpenAI Realtime API
BASE_URL: https://api.openai.com/v1
AUTH_HEADER: Authorization: Bearer {api_key}

ENDPOINTS:
  - method: WSS
    path: /v1/realtime?model={model}
    description: >
      Main WebSocket connection for real-time speech-to-speech and transcription
      sessions. Server-to-server usage with full API key. All realtime interaction
      (audio in/out, text, function calling, session management) happens via
      bidirectional JSON event messages over this socket. Optionally specify
      call_id instead of model when attaching a sideband control channel to
      an existing WebRTC or SIP call.
    request_format: >
      WebSocket upgrade (GET with Upgrade: websocket). After connect, send/receive
      JSON-serialized event objects. Audio payloads are Base64-encoded PCM16/G.711.
      Query params: model (e.g. gpt-realtime) OR call_id (for sideband attach).
      Auth header: Authorization: Bearer {api_key}.
    python_request_param: websocket (websocket-client library)
    response_format: >
      Stream of JSON server-event objects, e.g. session.created, response.audio.delta,
      conversation.item.created, rate_limits.updated, error, etc.

  - method: POST
    path: /v1/realtime/client_secrets
    description: >
      Generate a short-lived ephemeral API key (client secret) for use in
      browser/mobile WebRTC or WebSocket connections. Accepts full session
      configuration (model, voice, audio format, session type, etc.). Returns
      an ephemeral token string (ek_...) valid for one session.
    request_format: application/json body with session config object
    python_request_param: json=
    response_format: >
      JSON: { "value": "ek_...", "expires_at": <unix_timestamp> }

  - method: POST
    path: /v1/realtime/calls
    description: >
      WebRTC SDP offer negotiation. Submit an SDP offer body and receive an
      SDP answer. The response Location header contains a unique call_id that
      can be used to attach a server-side WebSocket sideband connection.
    request_format: >
      Content-Type: application/sdp. Body: SDP offer text. Auth via ephemeral key.
    python_request_param: data=
    response_format: >
      SDP answer body. Location header contains the call_id URL.

  - method: POST
    path: /v1/realtime/calls/{call_id}/accept
    description: >
      Accept an incoming SIP call identified by call_id. Configures the
      session for the accepted call.
    request_format: application/json (session config options)
    python_request_param: json=
    response_format: 200 OK JSON

  - method: POST
    path: /v1/realtime/calls/{call_id}/reject
    description: >
      Decline an incoming SIP call. Optional JSON body with status_code
      (e.g. 486 for busy). Defaults to 603 Decline.
    request_format: application/json { "status_code": <int> }
    python_request_param: json=
    response_format: 200 OK after OpenAI delivers SIP response

  - method: POST
    path: /v1/realtime/calls/{call_id}/hangup
    description: >
      Terminate an active SIP or WebRTC realtime call. Works for both call types.
    request_format: No body required
    python_request_param: (none — empty POST)
    response_format: 200 OK when teardown begins

  - method: POST
    path: /v1/realtime/calls/{call_id}/refer
    description: >
      Transfer an active SIP call to another endpoint. Provide target_uri
      (e.g. tel:+14155550123 or sip:agent@example.com) in JSON body.
    request_format: application/json { "target_uri": "<sip_or_tel_uri>" }
    python_request_param: json=
    response_format: 200 OK once REFER is relayed to SIP provider

RESPONSE_FORMAT: >
  REST endpoints: Standard JSON bodies. WebSocket: bidirectional stream of

SDK_PACKAGE: pip install openai  (official); pip install websocket-client (low-level WS)

  Text: UTF-8 JSON event strings over WebSocket.

PYTHON_EXAMPLES:
  # Example 1: Server-side WebSocket connection (voice agent)
  import os, json, websocket

  OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY")
  url = "wss://api.openai.com/v1/realtime?model=gpt-realtime"
  headers = ["Authorization: Bearer " + OPENAI_API_KEY]

  def on_open(ws):
      print("Connected to server.")
      ws.send(json.dumps({
          "type": "session.update",
          "session": {
              "type": "realtime",
              "instructions": "You are a helpful voice assistant.",
              "voice": "marin",
              "turn_detection": {"type": "server_vad"}
          }
      }))

  def on_message(ws, message):
      data = json.loads(message)
      print("Received event:", json.dumps(data, indent=2))

  ws = websocket.WebSocketApp(
      url, header=headers, on_open=on_open, on_message=on_message
  )
  ws.run_forever()

  # Example 2: Generate ephemeral client secret (for browser/WebRTC use)
  import requests, json, os

  api_key = os.environ["OPENAI_API_KEY"]
  response = requests.post(
      "https://api.openai.com/v1/realtime/client_secrets",
      headers={
          "Authorization": f"Bearer {api_key}",
          "Content-Type": "application/json"
      },
      json={
          "session": {
              "type": "realtime",
              "model": "gpt-realtime",
              "audio": {"output": {"voice": "marin"}}
          }
      }
  )
  ephemeral_key = response.json()["value"]
  print("Ephemeral key:", ephemeral_key)

  # Example 3: Hang up a SIP/WebRTC call
  import requests, os
  call_id = "rtc_u1_9c6574da8b8a41a18da9308f4ad974ce"
